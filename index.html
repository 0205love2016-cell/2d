const gvizUrl = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/gviz/tq?tqx=out:json&sheet=${encodeURIComponent(sheetName)}`;
const res = await fetch(gvizUrl);
const text = await res.text();
const jsonMatch = text.match(/google\.visualization\.Query\.setResponse\(([\s\S]+)\);/);
if (!jsonMatch) return null;
const data = JSON.parse(jsonMatch[1]);
if (!data || !data.table || !data.table.rows) return null;
const rows = data.table.rows;
let quotaMap = {};
let memberFixedLeaves = {};
let quotaRowIndex = -1;
for (let i = 0; i < rows.length; i++) {
const c = rows[i].c;
const label = c && c[0] && c[0].v ? String(c[0].v).trim() : "";
if (label.includes("可填") || label.includes("名額")) {
quotaRowIndex = i;
break;
}
}
if (quotaRowIndex === -1) {
for (let i = 0; i < rows.length; i++) {
const c = rows[i].c;
const label = c && c[0] && c[0].v ? String(c[0].v).trim() : "";
if (label.includes("人數")) {
quotaRowIndex = i;
break;
}
}
}
if (quotaRowIndex !== -1) {
const qRow = rows[quotaRowIndex].c;
for (let d = 1; d <= 31; d++) {
if (d < qRow.length && qRow[d] && qRow[d].v !== null && qRow[d].v !== undefined) {
const val = parseInt(qRow[d].v, 10);
const dateStr = `${year}-${monthStr}-${String(d).padStart(2, '0')}`;
quotaMap[dateStr] = isNaN(val) ? 0 : val;
}
}
}
for (let r = 0; r < rows.length; r++) {
const c = rows[r].c;
const rowName = c && c[0] && c[0].v ? String(c[0].v).trim() : "";
if (rowName && !rowName.includes("人數") && !rowName.includes("名額")) {
let leaves = {};
let hasLeave = false;
for (let d = 1; d <= 31; d++) {
if (d < c.length && c[d] && c[d].v) {
const v = String(c[d].v).trim();
if (v !== "") {
const dateStr = `${year}-${monthStr}-${String(d).padStart(2, '0')}`;
leaves[dateStr] = v;
hasLeave = true;
}
}
}
if (hasLeave) {
memberFixedLeaves[rowName] = leaves;
}
}
}
return {
sheetFound: true,
sheetName: sheetName,
quotaMap: quotaMap,
memberFixedLeaves: memberFixedLeaves
};
} catch (err) {
return null;
}
}

/**
* 等待 LIFF SDK 載入完成 (防止非同步競態)
*/
async function waitForLiffSdk(maxMs = 3000) {
const start = Date.now();
while (typeof liff === 'undefined' && (Date.now() - start < maxMs)) {
await new Promise(r => setTimeout(r, 100));
}
return typeof liff !== 'undefined';
}
const liffPromise = (async () => {
try {
const liffAvailable = await waitForLiffSdk(3000);
if (!liffAvailable) return null;
await liff.init({ liffId: LIFF_ID });
if (!liff.isLoggedIn()) {
liff.login();
return null;
}
const profile = await liff.getProfile();
return profile.userId;
} catch (e) {
return null;
}
})();
const [initData, freshUserId] = await Promise.all([apiPromise, liffPromise]);
let finalInitData = initData;
if (finalInitData && !finalInitData.error) {
// 🛡️ 純前端主動降級容錯：若 GAS 快取鎖死或找不到分頁，前端直接穿透 Google 試算表補足
if (!finalInitData.sheetFound || !finalInitData.quotaMap || Object.keys(finalInitData.quotaMap).length === 0) {
const fallback = await fetchDirectSheetFallback(state.targetYear, state.targetMonth);
if (fallback) {
finalInitData.sheetFound = true;
finalInitData.sheetName = fallback.sheetName;
finalInitData.quotaMap = fallback.quotaMap;
finalInitData.memberFixedLeaves = fallback.memberFixedLeaves;
}
}
if (freshUserId) {
state.lineId = freshUserId;
localStorage.setItem('last_user_line_id_xingang', freshUserId);
}
applyInitData(finalInitData, freshUserId || state.lineId);
} else {
// 若後端 API 逾時，嘗試前端直連試算表
const fallback = await fetchDirectSheetFallback(state.targetYear, state.targetMonth);
if (fallback) {
fallback.queueList = [];
fallback.personnelList = [];
fallback.recordsList = [];
applyInitData(fallback, freshUserId || state.lineId);
} else {
const syncStatus = document.getElementById('syncStatusBar');
if (syncStatus) {
syncStatus.className = "sync-status updating";
syncStatus.innerHTML = `<span style="color:#d32f2f;">⚠️ 連線較慢，請點此 <a href="javascript:location.reload()">🔄 重新整理</a></span>`;
}
}
}
}
/**
* 套用後端最新資料並平滑更新介面 (支援客戶端身分即時動態映射)
*/
function applyInitData(initData, activeLineId) {
latestInitData = initData;
const targetId = activeLineId || state.lineId;
let matchedName = initData.myName || state.name;
// 若後端有提供對照表，直接在客戶端進行身分映射，完全免去第二次網路請求
if (targetId && initData.lineIdToName && initData.lineIdToName[targetId]) {
matchedName = initData.lineIdToName[targetId];
}
state.name = matchedName;
if (state.name) localStorage.setItem('last_user_name_xingang', state.name);
state.allFinished = !!initData.allFinished;
state.currentActiveNames = initData.currentActiveNames || [];
state.currentActiveLineIds = initData.currentActiveLineIds || [];
// 客戶端即時計算是否輪到我與狀態訊息 (遵守新港分隊佇列與規則，支援同序號並行)
const isInQueue = initData.queueList && initData.queueList.some(item => item.name === matchedName);
const isUserActive = (targetId && initData.currentActiveLineIds && initData.currentActiveLineIds.includes(targetId)) ||
(matchedName && initData.currentActiveNames && initData.currentActiveNames.includes(matchedName));
if (initData.allFinished) {
state.isMyTurn = isInQueue;
state.statusMessage = isInQueue ? '🎉 自由換假模式：可退回或自選假！' : '👀 瀏覽模式：大家已排假完成。';
} else if (isUserActive) {
state.isMyTurn = true;
if (initData.currentActiveNames && initData.currentActiveNames.length > 1) {
const others = initData.currentActiveNames.filter(n => n !== matchedName);
state.statusMessage = `👉 輪到您填假了！(目前與 ${others.join('、')} 同序列填寫中)`;
} else {
state.statusMessage = '👉 輪到您填假了！';
}
} else if (initData.currentActiveNames && initData.currentActiveNames.length > 0) {
state.isMyTurn = false;
const namesStr = initData.currentActiveNames.join('、');
state.statusMessage = isInQueue ? ('⏳ 目前輪到 [' + namesStr + '] 填寫。') : ('👀 瀏覽模式 (目前輪到 [' + namesStr + '])');
} else {
state.isMyTurn = initData.isMyTurn;
state.statusMessage = initData.message;
}
if (!matchedName && targetId) {
state.statusMessage = '⚠️ 找不到您的系統 ID，請通知管理員。<a href="javascript:void(0)" onclick="toggleIdentitySelector()" style="color:#007bff; text-decoration:underline; font-size:13px; margin-left:6px;">[手動選身分]</a>';
}
// 客戶端計算個人已選假單 userBooked
if (matchedName && initData.recordsList) {
state.userBooked = [];
for (let i = 0; i < initData.recordsList.length; i++) {
if (initData.recordsList[i].name === matchedName) {
state.userBooked.push(initData.recordsList[i].date);
}
}
} else {
state.userBooked = initData.userBooked || [];
}
// 客戶端計算個人固定假 fixedLeaves (新港分隊)
if (matchedName && initData.memberFixedLeaves && initData.memberFixedLeaves[matchedName]) {
state.fixedLeaves = initData.memberFixedLeaves[matchedName];
} else {
state.fixedLeaves = initData.fixedLeaves || {};
}
state.queueList = initData.queueList || [];
state.personnelList = initData.personnelList || [];
state.quotaMap = initData.quotaMap || {};
state.sheetFound = initData.sheetFound;
state.sheetName = initData.sheetName;
if (initData.sheetFound) {
try {
localStorage.setItem(cacheKey, JSON.stringify(initData));
} catch (e) {}
} else {
try {
localStorage.removeItem(cacheKey); // 若找不到分頁，立即清除本機快取，防範被鎖死
} catch (e) {}
}
renderAppSkeleton("✅ 資料已同步最新", false);
}

let isBooked = el.classList.contains('user-booked');

return;
}
if (isBooked) el.classList.add('selected-repick');
else el.classList.add('selected');

function toggleQueue() {
const qDiv = document.getElementById('queueContainer');
qDiv.style.display = (qDiv.style.display === 'block') ? 'none' : 'block';
if (qDiv.style.display === 'block') {
if (state.queueList.length === 0) { qDiv.innerHTML = "目前無排隊資料"; return; }
qDiv.innerHTML = state.queueList.map((item, i) => {
let isDone = item.status === "已完成";
let isAbandoned = item.status === "已放棄";
let isNow = !isDone && !isAbandoned && state.currentActiveNames && state.currentActiveNames.includes(item.name);
let statusClass = isDone ? "status-done" : (isAbandoned ? "status-abandoned" : (isNow ? "status-now" : "status-wait"));
let icon = isDone ? "✅" : (isAbandoned ? "🚫" : (isNow ? "👉" : "⏳"));
let label = isDone ? "已完成" : (isAbandoned ? "已放棄" : (isNow ? "填寫中" : "等待中"));
let roundPrefix = item.round ? `【${item.round}】` : '';
let orderPrefix = item.order ? `序號 ${item.order} ` : '';
return `<div class="queue-item ${statusClass}"><span>${roundPrefix}${orderPrefix}${item.name}</span> <span>${icon} ${label}</span></div>`;
}).join('');
}
}
